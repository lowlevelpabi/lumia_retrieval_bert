Table: papers
CREATE TABLE papers (
	id INTEGER NOT NULL, 
	title VARCHAR, 
	author VARCHAR, 
	year VARCHAR, 
	abstract TEXT, 
	department VARCHAR, 
	keywords TEXT, 
	project_type VARCHAR, 
	degree_program VARCHAR, 
	citation_count INTEGER, 
	view_count INTEGER, 
	file_path VARCHAR, 
	created_at DATETIME, 
	updated_at DATETIME, 
	PRIMARY KEY (id)
)

Table: user_citations
CREATE TABLE user_citations (
	id INTEGER NOT NULL, 
	user_id INTEGER NOT NULL, 
	paper_id INTEGER NOT NULL, 
	created_at DATETIME, 
	PRIMARY KEY (id), 
	CONSTRAINT unique_user_paper_citation UNIQUE (user_id, paper_id), 
	FOREIGN KEY(user_id) REFERENCES users (id) ON DELETE CASCADE, 
	FOREIGN KEY(paper_id) REFERENCES papers (id) ON DELETE CASCADE
)
----------------------------------------
Columns:
  (0, 'id', 'INTEGER', 1, None, 1)
  (1, 'user_id', 'INTEGER', 1, None, 0)
  (2, 'paper_id', 'INTEGER', 1, None, 0)
  (3, 'created_at', 'DATETIME', 0, None, 0)

Table: borrow_records
CREATE TABLE borrow_records (
	id INTEGER NOT NULL, 
	paper_id INTEGER, 
	user_id INTEGER, 
	borrow_date DATETIME, 
	due_date DATETIME, 
	return_date DATETIME, 
	status VARCHAR, 
	PRIMARY KEY (id),

	FOREIGN KEY(user_id) REFERENCES users (id)
